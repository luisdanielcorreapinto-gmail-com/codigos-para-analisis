ALTER PROCEDURE [dbo].[spStockInquiryConc] (
	@Workstation		VARCHAR(10),
	@UserNo				INTEGER,
	@FromDate 			DATETIME,	--The date range from which the report should draw data	
 	@ToDate				DATETIME,
	@Location			VARCHAR(4),	--Stock Location	
	@UseStartTimeOfDay	VARCHAR(1),
	@CostType			VARCHAR(10) = 'STANDARD',	-- Pass the 
	@SQLFilter			VARCHAR(5000) = '',
  @DebugMode CHAR(1) = 'N'
	)
AS
/****************************************************************************************************************/
/*	spStockInquiryConc VERSION 3.00.03(M) 07/03/2005								*/
/****************************************************************************************************************/
-- Prevent any shared locks being created during execution of SP
SET TRANSACTION ISOLATION LEVEL READ UNCOMMITTED
DECLARE @InquiryID		AS INTEGER
DECLARE	@AppendExisting 	AS CHAR(1)
DECLARE	@LastTransaction	AS DATETIME 
DECLARE @SQL			as NVARCHAR(MAX)
DECLARE @SelectionStart		AS DATETIME
DECLARE @FromDateTime		AS DATETIME
DECLARE @ToDateTime		AS DATETIME
DECLARE @StartTimeOfDay		AS DATETIME
DECLARE @MinLastStocktakeDate	AS DATETIME
DECLARE @TotalSales 		AS DECIMAL(28,13)
DECLARE @HighestStocktakeFrequency AS Integer
DECLARE @CostTax		AS VARCHAR(20)
DECLARE @RetailTax	AS VARCHAR(20)
DECLARE @UseFromDate as CHAR(1)
DECLARE @UseFromDateAsLastStocktakeDate as CHAR(1)
DECLARE @PriceBookToUse AS NVARCHAR(510)				
    
-- Variables for Debugging
DECLARE @StartTime as DateTime
-- Set the Value for Debug Mode 
SET NOCOUNT ON
SET @StartTime = dbo.GetLocalDate()
DECLARE @StocktakeMode		AS VARCHAR(30)
SELECT @StocktakeMode = Configure_strValue FROM tblConfigure WHERE Configure_strName = 'StocktakeMode'
DECLARE @CurrencyScale	INTEGER
SELECT @CurrencyScale = Configure_strValue FROM tblConfigure WHERE Configure_strName = 'CurrencyScale'
SET @PriceBookToUse = ISNULL( ( SELECT Configure_strValue FROM tblConfigure WHERE Configure_strName = 'PriceBookToUse' ), '' )
-- MP 2008/06/19
-- Default this to N, meaning the Last Stocktake Date is the actual Last Stocktake date for each Item
SET @UseFromDateAsLastStocktakeDate = 'N'
-- SC: 30 May. Use CurrencyFormat to determine the rounding place for Retail Price
DECLARE	@strRoundingSalesValue varchar(50)
DECLARE 	@intRoundingSalesValue int	
SELECT @strRoundingSalesValue =  Configure_strValue FROM tblConfigure WHERE Configure_strName = 'CurrencyFormat'
IF CHARINDEX('.', @strRoundingSalesValue)>0 
SET @intRoundingSalesValue = 
 LEN(RIGHT(@strRoundingSalesValue,LEN(@strRoundingSalesValue)-CHARINDEX('.', @strRoundingSalesValue)))
ELSE
SET  @intRoundingSalesValue = 0 	
DECLARE @CostCurrencyScale	AS INTEGER
SET @CostCurrencyScale = 10	--Hardcode the 10 decimal places because we should not actually be doing any rounding on Cost
-- Retail Prices Include Tax
SELECT @RetailTax = Configure_strValue FROM tblConfigure WHERE Configure_strName = 'PriceModeConcessions'
-- Cost Prices Include Tax
SELECT @CostTax = Configure_strValue FROM tblConfigure WHERE Configure_strName = 'CostModeConcessions'
-- Validate what information was passed into this SP
--
--	B16135 MTS :: Documentation
--
--	Why is this segment of code here?
--		When the Stock Inquiry Conc[essions] SP is executed from "create stock take"
--		it passes a NULL value as @FromDate, so the first test fails. It also passes
--		empty string as @Location, so the first test in the ELSE block succeeds and
--		the variable @UseFromDateAsLastStocktakeDate is set to Y. This variable is
--		used to overwrite ALL the LastStocktakeDate values immediately before the
--		end of the SP with the value of @FromDate.
--
--		B16135 removes the "at all costs" over-write under instruction from Mark Pattie.
--		This change may cause future issues because the behaviour has been changed.
--
IF IsDate(@FromDate) <> 1 AND NOT ISNULL(@SQLFilter, '') = ''
BEGIN 
		SET @SQL = 
			N'SELECT @LastSTake = MAX(Stock_dtmLastSTake) 
				FROM (SELECT ss.*, 
							i.Item_strItemId AS Item_strCode, i.Class_strCode AS Item_strClass, i.StakeGrp_strCode, i.Vendor_strCode, g.StakeGrp_strFrequency AS StocktakeFrequencyCode
						FROM tblStock_Status ss
						JOIN tblItem i ON i.Item_strItemId = ss.Item_strItemId
						LEFT JOIN tblStocktake_Group g ON g.StakeGrp_strCode = i.StakeGrp_strCode) a
				WHERE 1 = 1 ' + @SQLFilter 
				
		EXEC sp_executesql @SQL, 
	           N'@LastSTake DATETIME output', @LastSTake = @FromDate output;			
		IF IsDate(@FromDate) = 1
		BEGIN
			SET @UseFromDate = 'Y'
			
			-- MP 2008/06/19
			-- Change to Y, meaning the Last Stocktake Date now becomes the From Date - DONE at the end of this SP
			SET @UseFromDateAsLastStocktakeDate = 'Y'
		END
END
IF IsDate(@FromDate) = 1
	SET @UseFromDate = 'Y'
ELSE
BEGIN
	-- MP 2008/06/19
	-- If no From Date is used, then this must be a Stocktake Creation or Refresh. We will assign a date which is the Max Date of the last stocktake.
	-- If no Date Exists, then the @UseFromDate field will remain set to N
	-- Doing this because it is faster to provide a From Date and a To Date, rather than just a from Date
	-- Also need to manipulate the Last Stocktake Date field at the end of this SP to be the From Date Passed in, as the Opening Balance we calculate will be for the From Date and not the Actual Last Stocktake Date.
	IF ISNULL(@Location, '') = '' 
	BEGIN
		IF  (SELECT MAX(STake_dtmDateTime) FROM tblStockTakeDates) IS NOT NULL 
		BEGIN
			SET @FromDate = (SELECT MAX(STake_dtmDateTime) FROM tblStockTakeDates)
			SET @UseFromDate = 'Y'
			
			-- MP 2008/06/19
			-- Change to Y, meaning the Last Stocktake Date now becomes the From Date - DONE at the end of this SP
			SET @UseFromDateAsLastStocktakeDate = 'Y'
		END
		ELSE
		BEGIN
			SET @UseFromDate = 'N'
		END
	END
	ELSE
	BEGIN	
		IF  (SELECT MAX(STake_dtmDateTime) FROM tblStockTakeDates WHERE Location_strCode = @Location) IS NOT NULL 
		BEGIN
			SET @FromDate = (SELECT MAX(STake_dtmDateTime) FROM tblStockTakeDates WHERE Location_strCode = @Location)
			SET @UseFromDate = 'Y'
			-- Change to Y, meaning the Last Stocktake Date now becomes the From Date - DONE at the end of this SP
			SET @UseFromDateAsLastStocktakeDate = 'Y'
		END
		ELSE
		BEGIN
			SET @UseFromDate = 'N'
		END
	END			 
END
IF @UseStartTimeOfDay = 'Y'
	BEGIN
		SELECT @StartTimeOfDay = Configure_strValue From tblConfigure 
		Where Configure_strName = 'StartTimeOfDay'
		SELECT @FromDateTime = DATEADD(hh, datepart(hh,@StartTimeOfDay), @FromDate)
		SELECT @FromDateTime = DATEADD(mi, datepart(mi,@StartTimeOfDay), @FromDateTime)
		SELECT @ToDateTime = DATEADD(hh, datepart(hh,@StartTimeOfDay), @ToDate)
		SELECT @ToDateTime = DATEADD(mi, datepart(mi,@StartTimeOfDay), @ToDateTime)
	END
ELSE
	BEGIN
		SET @FromDateTime = @FromDate
		SET @ToDateTime = @ToDate
	END
	SELECT @InquiryID = Inquiry_intID FROM tblStockInquiryConcHeader
	WHERE WorkStation_strCode = @Workstation
	AND User_intUserNo = @UserNo
--------------------------------------------------------------------------------------------------------------
-- If the Opening Date passed in is Null, then set the From Date to be the Closing Date
-- The reason is because the FromDateTime is used to find the stocktake prior to this date.
-- If it is null, then the SP wont find a previous stocktake.  So set it to the closing date, because
-- in this case, we want to find the stocktake that occured prior to the closing date
If @UseFromDate = 'N' 
	SET @FromDateTime = @ToDateTime
--------------------------------------------------------------------------------------------------------------
IF @DebugMode = 'Y'
BEGIN
	PRINT @StartTime
END
DECLARE @CurrentDate DateTime
SET @CurrentDate = dbo.GetLocalDate()
--	B16135 MTS : @InquiryID is ALWAYS NULL at this point, so why is there
--				 an IF...ELSE construct?
IF @InquiryID IS NULL 
BEGIN
	-- Delete data in these tables more than 2 hours old. (MP 23/6/08)
	DELETE FROM tblStockInquiryConcHeader WHERE RunDateTime < DATEADD(hh, -2, @CurrentDate)
	DELETE FROM tblStockInquiryConc WHERE Inquiry_intID NOT IN (SELECT DISTINCT Inquiry_intID FROM tblStockInquiryConcHeader)
	SET @AppendExisting = 'N'
END
ELSE
BEGIN
	IF (SELECT DateFrom FROM tblStockInquiryConcHeader WHERE Inquiry_intID = @InquiryID) = @FromDate
	AND (SELECT DateTo FROM tblStockInquiryConcHeader WHERE Inquiry_intID = @InquiryID) = @ToDate
	AND (SELECT Location_strCode FROM tblStockInquiryConcHeader WHERE Inquiry_intID = @InquiryID) = @Location
	BEGIN
		SET @AppendExisting = 'Y'		
		SELECT @LastTransaction = LastTransDateTime FROM tblStockInquiryConcHeader
		WHERE Inquiry_intID = @InquiryID
	END
	ELSE
	BEGIN
		DELETE FROM tblStockInquiryConc WHERE Inquiry_intID = @InquiryID
		DELETE FROM tblStockInquiryConcHeader WHERE Inquiry_intID = @InquiryID
		SET @AppendExisting = 'N'
		-- Delete data in these tables more than 2 hours old. (MP 23/6/08)
		DELETE FROM tblStockInquiryConcHeader WHERE RunDateTime < DATEADD(hh, -2, @CurrentDate)
		DELETE FROM tblStockInquiryConc WHERE Inquiry_intID NOT IN (SELECT DISTINCT Inquiry_intID FROM tblStockInquiryConcHeader)
		
	END
END
IF @AppendExisting = 'N'
BEGIN
	INSERT INTO tblStockInquiryConcHeader 
	(User_intUserNo,WorkStation_strCode,DateFrom,DateTo,Location_strCode,RunDateTime,AppendExisting, CostType)
	VALUES	
	(@UserNo,@Workstation,@FromDate,@ToDate,@Location,dbo.GetLocalDate(),'N', @CostType)
	SELECT @InquiryID = Inquiry_intID FROM tblStockInquiryConcHeader
	WHERE WorkStation_strCode = @Workstation
	AND User_intUserNo = @UserNo
	DECLARE @ActualDate datetime = ISNULL(@ToDate, dbo.GetLocalDate());
	WITH ItemPriceListCTE AS
	(
		--Replace it if there is a valid price in the price book.
		SELECT
			D.PriceD_curPrice, 
			D.Item_strItemId,
			ROW_NUMBER() OVER 
			(
				PARTITION BY
					PL.CinOperator_strCode,
					D.Item_strItemId
				ORDER BY 
					CASE WHEN @PriceBookToUse = 'Most Recent Start Date' THEN CAST(H.PriceH_dtmEffectiveFrom AS date) ELSE NULL END DESC,
					CASE WHEN @PriceBookToUse = 'Most Recent Start Date' THEN H.PriceH_intStartTime ELSE NULL END DESC,
					D.PriceD_curPrice ASC
			) AS [Rank]
		FROM 
			tblPriceListHeader H
		INNER JOIN 
			tblPriceListDetails D
		ON
			D.PriceH_strCode = H.PriceH_strCode
		AND
			D.PriceD_strStatus = 'A'
		INNER JOIN
			tblStock_Status SS 
		ON
			SS.Item_strItemId = D.Item_strItemId
		INNER JOIN 
			tblStock_Location SL
		ON
			SL.Location_strCode = SS.Location_strCode
		AND
			SL.Location_strDefault = 'Y'
		INNER JOIN 
			tblPriceListLink PL
		ON 
			PL.PriceH_strCode = H.PriceH_strCode
		AND 
			PL.CinOperator_strCode = SL.CinOperator_strCode
		WHERE
			H.PriceH_strStatus = 'A'
		AND 
			H.PriceH_dtmEffectiveFrom <= @ActualDate
		AND 
			H.PriceH_dtmEffectiveTo >= @ActualDate
		AND 
			CASE DATEPART(dw, @ActualDate)
				WHEN 1 THEN H.PriceH_strSunday
				WHEN 2 THEN H.PriceH_strMonday
				WHEN 3 THEN H.PriceH_strTuesday
				WHEN 4 THEN H.PriceH_strWednesday
				WHEN 5 THEN H.PriceH_strThursday
				WHEN 6 THEN H.PriceH_strFriday
				WHEN 7 THEN H.PriceH_strSaturday 
			END = 'Y'
		-- MVW 22 Jul 2015  Default to Price Book with Latest Effective From Date
	)
	-- MP 2007/11/15 V3R3 Change to only include Item/Location Combinations that are in Stock Status.
	INSERT INTO tblStockInquiryConc
	(	Inquiry_intID,Item_strCode,Item_strClass,Location_strCode,Location_strDescription,Class_strDescription,
		Item_strItemType,Item_strReport,Item_strDescription, BaseUOMCode,StockUOMCode,StockUOMConv, SaleUOMConv,
		Item_strStocktakeable, StakeGrp_strCode, Item_strBoxOfficeSundry, Item_strHOCode, Item_strStatus, Vendor_strCode,
		Item_curRetailPrice, BusinessDate, Item_decCostPrice
	
	)
	
	--MNS 22 June 2012 T20510: Use fnGetStockRetailPrice to find the items retail price, instead of just selecting it from tblItem.
  --WW 14 May 2018: move logic in fnGetStockRetailPrice as a SQL JOIN, because it was slowing down the stored procedure by 10+ seconds one some databases.
	SELECT @InquiryID, i.Item_strItemId,i.Class_strCode,sl.Location_strCode, sl.Location_strDescription, 
		c.Class_strDescription, i.Item_strItemType,'N', i.Item_strItemDescription, i.Item_strBaseUOMCode, 
		i.Item_strStockUOMCode, i.Item_curStockUOMConv, i.Item_curSaleUOMConv,
		i.Item_strStocktakeable, i.StakeGrp_strCode, i.Item_strReport, i.Item_strMasterItemCode, i.Item_strStatus, i.Vendor_strCode,
		COALESCE(BestPrice.PriceD_curPrice,i.Item_curRetailPrice, 0), @ToDate, i.Item_decCostPrice
	FROM tblItem i
  INNER JOIN tblItem_Class c ON i.Class_strCode = c.Class_strCode
  INNER JOIN tblStock_Status status ON i.Item_strItemId = status.Item_strItemId
  INNER JOIN tblStock_Location sl ON sl.Location_strCode = status.Location_strCode
  LEFT JOIN ItemPriceListCTE As BestPrice ON BestPrice.Item_strItemId = i.Item_strItemId AND BestPrice.Rank = 1
	ORDER BY i.Item_strItemId, sl.Location_strCode
  OPTION(RECOMPILE)
	-- If in Debug Mode, then Print out Debugging info
	IF @DebugMode = 'Y'
	BEGIN
		PRINT 'Get all Items took: ' + CAST(DATEDIFF(ms, @StartTime, dbo.GetLocalDate()) as VarChar(30)) + ' ms'
		SET @StartTime = dbo.GetLocalDate()
	END
	IF @Location <> ''
		DELETE FROM tblStockInquiryConc WHERE Location_strCode <> @Location
		AND Inquiry_intID = @InquiryID
	UPDATE tblStockInquiryConc
	SET Vendor_strDescription = v.Vendor_strName
	FROM tblVendor v, tblStockInquiryConc sc
	WHERE sc.Vendor_strCode = v.Vendor_strCode
	AND sc.Inquiry_intID = @InquiryID
	-- Set the Retail Price to be what ever the system setting is. If TaxINCLUSIVE, then nothing gets changed,
	-- If tax exclusive, we need to take the tax off.
	
	
If @RetailTax = 'TaxINCLUSIVE'	-- If Prices include tax, then for Net we take it off
	BEGIN
		UPDATE tblStockInquiryConc
		SET Item_curNetRetailPrice = sc.Item_curRetailPrice - Round(CASE WHEN sc.Item_curRetailPrice < (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
										CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
										CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
										CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END) THEN Round(sc.Item_curRetailPrice, @intRoundingSalesValue)
	
									ELSE	(CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate,@intRoundingSalesValue) ELSE 0 END + 
									CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
									CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
									CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END) +
	 								
									CASE WHEN tax.STax_strRateType = 'P' Then Round(sc.Item_curRetailPrice - Round((CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END), @intRoundingSalesValue) - (sc.Item_curRetailPrice - (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END)) / ((tax.STax_curRate/100) + 1), @intRoundingSalesValue) Else 0 END + 
	
		 							CASE WHEN tax.STax_strRateType2 = 'P' Then Round(sc.Item_curRetailPrice - (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END), @intRoundingSalesValue) - Round((sc.Item_curRetailPrice - (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END)) / ((tax.STax_curRate2/100) + 1), @intRoundingSalesValue) Else 0 END + 
	
		 							CASE WHEN tax.STax_strRateType3 = 'P' Then Round(sc.Item_curRetailPrice - (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END), @intRoundingSalesValue) - Round((sc.Item_curRetailPrice - (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END)) / ((tax.STax_curRate3/100) + 1), @intRoundingSalesValue) Else 0 END + 
	
								 	CASE WHEN tax.STax_strRateType4 = 'P' Then Round(sc.Item_curRetailPrice - (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END), @intRoundingSalesValue) - Round((sc.Item_curRetailPrice - (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END)) / ((tax.STax_curRate4/100) + 1), @intRoundingSalesValue) Else 0 END  
	
									END, @intRoundingSalesValue)
		FROM tblStockInquiryConc sc, tblItem item, tblSalesTax tax
		WHERE sc.Item_strCode = item.Item_strItemId
		AND item.STax_strCode = tax.STax_strCode
		AND sc.Inquiry_intID = @InquiryID
	END
	ELSE	-- If Prices Exclude Tax, then Retail Price becomes Net Price, and then add tax to retail figure
	BEGIN
		
		UPDATE tblStockInquiryConc
		SET Item_curNetRetailPrice = sc.Item_curRetailPrice,
		    Item_curRetailPrice = Round(sc.Item_curRetailPrice + (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
									CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
									CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
									CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END) +
	 								Round(CASE WHEN tax.STax_strRateType = 'P' Then (tax.STax_curRate/100) * (sc.Item_curRetailPrice + (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END))Else 0 END, @intRoundingSalesValue) + 
	
		 							Round(CASE WHEN tax.STax_strRateType2 = 'P' Then (tax.STax_curRate2/100) * (sc.Item_curRetailPrice + (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END))Else 0 END, @intRoundingSalesValue) + 
	
		 							Round(CASE WHEN tax.STax_strRateType3 = 'P' Then (tax.STax_curRate3/100) * (sc.Item_curRetailPrice + (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END))Else 0 END, @intRoundingSalesValue) + 
	
								 	Round(CASE WHEN tax.STax_strRateType4 = 'P' Then (tax.STax_curRate4/100) * (sc.Item_curRetailPrice + (CASE WHEN tax.STax_strRateType = 'F' Then Round(tax.STax_curRate, @intRoundingSalesValue) ELSE 0 END + 
																					CASE WHEN tax.STax_strRateType2 = 'F' Then Round(tax.STax_curRate2, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType3 = 'F' Then Round(tax.STax_curRate3, @intRoundingSalesValue) ELSE 0 END +
																					CASE WHEN tax.STax_strRateType4 = 'F' Then Round(tax.STax_curRate4, @intRoundingSalesValue) ELSE 0 END))Else 0 END, @intRoundingSalesValue), @intRoundingSalesValue)
		FROM tblStockInquiryConc sc, tblItem item, tblSalesTax tax
		WHERE sc.Item_strCode = item.Item_strItemId
